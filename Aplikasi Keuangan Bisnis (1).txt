import React, { useState, useEffect, useMemo } from 'react';
import { 
  LayoutDashboard, 
  Receipt, 
  PieChart, 
  TrendingUp, 
  TrendingDown, 
  Wallet, 
  Plus, 
  Search, 
  Edit, 
  Trash2, 
  Save, 
  X,
  AlertCircle
} from 'lucide-react';

// --- UTILITIES ---
const formatIDR = (amount) => {
  return new Intl.NumberFormat('id-ID', {
    style: 'currency',
    currency: 'IDR',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  }).format(amount);
};

const parseIDR = (value) => {
  return parseInt(value.toString().replace(/[^0-9]/g, ''), 10) || 0;
};

const formatInputRupiah = (value) => {
  const number = parseIDR(value);
  return number === 0 ? '' : number.toLocaleString('id-ID');
};

const CATEGORIES = {
  pemasukan: ['Penjualan', 'Pendapatan Jasa', 'Investasi', 'Lain-lain'],
  pengeluaran: ['Operasional', 'Bahan Baku', 'Gaji Karyawan', 'Pemasaran', 'Utang', 'Lain-lain']
};

// --- COMPONENTS ---

// 1. Dashboard Component
const Dashboard = ({ transactions }) => {
  const currentMonth = new Date().toISOString().slice(0, 7); // Format: YYYY-MM
  
  const metrics = useMemo(() => {
    let income = 0;
    let expense = 0;

    transactions.forEach(t => {
      // Filter untuk periode berjalan (bulan ini)
      if (t.date.startsWith(currentMonth)) {
        if (t.type === 'pemasukan') income += t.amount;
        if (t.type === 'pengeluaran') expense += t.amount;
      }
    });

    return {
      income,
      expense,
      balance: income - expense
    };
  }, [transactions, currentMonth]);

  const monthName = new Date().toLocaleDateString('id-ID', { month: 'long', year: 'numeric' });

  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
        <div>
          <h2 className="text-2xl font-bold text-slate-800">Ringkasan Finansial</h2>
          <p className="text-slate-500">Periode Berjalan: {monthName}</p>
        </div>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        {/* Total Pemasukan */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 flex items-center gap-4">
          <div className="p-4 bg-emerald-50 rounded-xl text-emerald-600">
            <TrendingUp size={32} />
          </div>
          <div>
            <p className="text-sm font-medium text-slate-500 mb-1">Total Pemasukan</p>
            <h3 className="text-2xl font-bold text-slate-800">{formatIDR(metrics.income)}</h3>
          </div>
        </div>

        {/* Total Pengeluaran */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 flex items-center gap-4">
          <div className="p-4 bg-rose-50 rounded-xl text-rose-600">
            <TrendingDown size={32} />
          </div>
          <div>
            <p className="text-sm font-medium text-slate-500 mb-1">Total Pengeluaran</p>
            <h3 className="text-2xl font-bold text-slate-800">{formatIDR(metrics.expense)}</h3>
          </div>
        </div>

        {/* Saldo Bersih */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 flex items-center gap-4">
          <div className={`p-4 rounded-xl ${metrics.balance >= 0 ? 'bg-blue-50 text-blue-600' : 'bg-orange-50 text-orange-600'}`}>
            <Wallet size={32} />
          </div>
          <div>
            <p className="text-sm font-medium text-slate-500 mb-1">Saldo Bersih</p>
            <h3 className={`text-2xl font-bold ${metrics.balance >= 0 ? 'text-slate-800' : 'text-orange-600'}`}>
              {formatIDR(metrics.balance)}
            </h3>
          </div>
        </div>
      </div>
    </div>
  );
};

// 2. Transaction Form Component
const TransactionForm = ({ onSave, initialData, onCancel }) => {
  const [formData, setFormData] = useState(initialData || {
    date: new Date().toISOString().split('T')[0],
    type: 'pemasukan',
    category: CATEGORIES.pemasukan[0],
    amount: '',
    notes: ''
  });

  const handleChange = (e) => {
    const { name, value } = e.target;
    
    if (name === 'type') {
      setFormData({
        ...formData,
        type: value,
        category: CATEGORIES[value][0] // Reset kategory saat tipe berubah
      });
    } else if (name === 'amount') {
      // Format input text to Rupiah dynamically
      setFormData({ ...formData, amount: formatInputRupiah(value) });
    } else {
      setFormData({ ...formData, [name]: value });
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.amount || parseIDR(formData.amount) <= 0) return;

    onSave({
      ...formData,
      amount: parseIDR(formData.amount)
    });
  };

  return (
    <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 mb-6">
      <div className="flex justify-between items-center mb-6">
        <h3 className="text-xl font-bold text-slate-800">
          {initialData ? 'Edit Transaksi' : 'Tambah Transaksi Baru'}
        </h3>
        {onCancel && (
          <button onClick={onCancel} className="p-2 hover:bg-slate-100 rounded-full text-slate-500 transition-colors">
            <X size={20} />
          </button>
        )}
      </div>

      <form onSubmit={handleSubmit} className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        <div className="space-y-2">
          <label className="text-sm font-medium text-slate-700">Tanggal</label>
          <input 
            type="date" 
            name="date"
            value={formData.date}
            onChange={handleChange}
            required
            className="w-full px-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
          />
        </div>

        <div className="space-y-2">
          <label className="text-sm font-medium text-slate-700">Jenis Transaksi</label>
          <select 
            name="type"
            value={formData.type}
            onChange={handleChange}
            className="w-full px-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
          >
            <option value="pemasukan">Pemasukan</option>
            <option value="pengeluaran">Pengeluaran</option>
          </select>
        </div>

        <div className="space-y-2">
          <label className="text-sm font-medium text-slate-700">Kategori</label>
          <select 
            name="category"
            value={formData.category}
            onChange={handleChange}
            className="w-full px-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
          >
            {CATEGORIES[formData.type].map(cat => (
              <option key={cat} value={cat}>{cat}</option>
            ))}
          </select>
        </div>

        <div className="space-y-2 lg:col-span-1">
          <label className="text-sm font-medium text-slate-700">Nominal (Rp)</label>
          <div className="relative">
            <span className="absolute left-4 top-2.5 text-slate-500 font-medium">Rp</span>
            <input 
              type="text" 
              name="amount"
              value={formData.amount}
              onChange={handleChange}
              placeholder="0"
              required
              className="w-full pl-12 pr-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
            />
          </div>
        </div>

        <div className="space-y-2 lg:col-span-2">
          <label className="text-sm font-medium text-slate-700">Catatan</label>
          <input 
            type="text" 
            name="notes"
            value={formData.notes}
            onChange={handleChange}
            placeholder="Deskripsi singkat..."
            required
            className="w-full px-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
          />
        </div>

        <div className="md:col-span-2 lg:col-span-3 flex justify-end gap-3 pt-2">
          {onCancel && (
            <button 
              type="button" 
              onClick={onCancel}
              className="px-6 py-2 border border-slate-200 text-slate-600 font-medium rounded-lg hover:bg-slate-50 transition-colors"
            >
              Batal
            </button>
          )}
          <button 
            type="submit"
            className="px-6 py-2 bg-blue-600 hover:bg-blue-700 text-white font-medium rounded-lg flex items-center gap-2 transition-colors shadow-sm"
          >
            <Save size={18} />
            Simpan Transaksi
          </button>
        </div>
      </form>
    </div>
  );
};

// 3. Transaction History Component
const TransactionManager = ({ transactions, setTransactions }) => {
  const [showForm, setShowForm] = useState(false);
  const [editingData, setEditingData] = useState(null);
  
  const [searchTerm, setSearchTerm] = useState('');
  const [filterMonth, setFilterMonth] = useState(new Date().toISOString().slice(0, 7));
  const [deleteConfirmId, setDeleteConfirmId] = useState(null);

  const handleSave = (data) => {
    if (editingData) {
      setTransactions(transactions.map(t => t.id === editingData.id ? { ...data, id: editingData.id } : t));
      setEditingData(null);
    } else {
      setTransactions([{ ...data, id: Date.now().toString() }, ...transactions]);
    }
    setShowForm(false);
  };

  const handleEdit = (transaction) => {
    setEditingData({ ...transaction, amount: transaction.amount.toLocaleString('id-ID') });
    setShowForm(true);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const handleDelete = (id) => {
    setTransactions(transactions.filter(t => t.id !== id));
    setDeleteConfirmId(null);
  };

  const filteredTransactions = transactions.filter(t => {
    const matchMonth = t.date.startsWith(filterMonth);
    const searchLower = searchTerm.toLowerCase();
    const matchSearch = t.notes.toLowerCase().includes(searchLower) || t.category.toLowerCase().includes(searchLower);
    return matchMonth && matchSearch;
  });

  return (
    <div>
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 mb-6">
        <h2 className="text-2xl font-bold text-slate-800">Manajemen Transaksi</h2>
        {!showForm && (
          <button 
            onClick={() => { setEditingData(null); setShowForm(true); }}
            className="bg-blue-600 hover:bg-blue-700 text-white px-5 py-2.5 rounded-lg flex items-center gap-2 font-medium transition-colors shadow-sm"
          >
            <Plus size={20} /> Tambah Data
          </button>
        )}
      </div>

      {showForm && (
        <TransactionForm 
          onSave={handleSave} 
          initialData={editingData} 
          onCancel={() => { setShowForm(false); setEditingData(null); }} 
        />
      )}

      {/* Filters & Search */}
      <div className="bg-white p-4 rounded-xl shadow-sm border border-slate-100 mb-6 flex flex-col md:flex-row gap-4">
        <div className="flex-1 relative">
          <Search size={18} className="absolute left-3 top-3 text-slate-400" />
          <input 
            type="text" 
            placeholder="Cari berdasarkan catatan atau kategori..."
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
            className="w-full pl-10 pr-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none"
          />
        </div>
        <div className="w-full md:w-48">
          <input 
            type="month" 
            value={filterMonth}
            onChange={(e) => setFilterMonth(e.target.value)}
            className="w-full px-4 py-2 border border-slate-200 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none"
          />
        </div>
      </div>

      {/* Table */}
      <div className="bg-white rounded-2xl shadow-sm border border-slate-100 overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse">
            <thead>
              <tr className="bg-slate-50 border-b border-slate-100 text-slate-600 text-sm font-semibold">
                <th className="p-4">Tanggal</th>
                <th className="p-4">Kategori & Catatan</th>
                <th className="p-4">Jenis</th>
                <th className="p-4 text-right">Nominal</th>
                <th className="p-4 text-center">Aksi</th>
              </tr>
            </thead>
            <tbody>
              {filteredTransactions.length === 0 ? (
                <tr>
                  <td colSpan="5" className="p-8 text-center text-slate-500">
                    <div className="flex flex-col items-center gap-2">
                      <Receipt size={40} className="text-slate-300" />
                      <p>Tidak ada transaksi ditemukan pada periode ini.</p>
                    </div>
                  </td>
                </tr>
              ) : (
                filteredTransactions.sort((a, b) => new Date(b.date) - new Date(a.date)).map(t => (
                  <tr key={t.id} className="border-b border-slate-50 hover:bg-slate-50/50 transition-colors">
                    <td className="p-4 text-slate-700 whitespace-nowrap">
                      {new Date(t.date).toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })}
                    </td>
                    <td className="p-4">
                      <p className="font-medium text-slate-800">{t.notes}</p>
                      <span className="inline-block mt-1 text-xs px-2 py-1 bg-slate-100 text-slate-600 rounded-md">
                        {t.category}
                      </span>
                    </td>
                    <td className="p-4">
                      <span className={`inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold ${
                        t.type === 'pemasukan' ? 'bg-emerald-50 text-emerald-600' : 'bg-rose-50 text-rose-600'
                      }`}>
                        {t.type === 'pemasukan' ? <TrendingUp size={12} /> : <TrendingDown size={12} />}
                        {t.type.charAt(0).toUpperCase() + t.type.slice(1)}
                      </span>
                    </td>
                    <td className={`p-4 text-right font-bold whitespace-nowrap ${
                      t.type === 'pemasukan' ? 'text-emerald-600' : 'text-slate-800'
                    }`}>
                      {t.type === 'pemasukan' ? '+' : '-'}{formatIDR(t.amount)}
                    </td>
                    <td className="p-4">
                      {deleteConfirmId === t.id ? (
                        <div className="flex justify-center items-center gap-2">
                          <button onClick={() => handleDelete(t.id)} className="text-xs bg-rose-600 text-white px-3 py-1.5 rounded hover:bg-rose-700">Hapus</button>
                          <button onClick={() => setDeleteConfirmId(null)} className="text-xs bg-slate-200 text-slate-700 px-3 py-1.5 rounded hover:bg-slate-300">Batal</button>
                        </div>
                      ) : (
                        <div className="flex justify-center items-center gap-2">
                          <button 
                            onClick={() => handleEdit(t)}
                            className="p-2 text-slate-400 hover:text-blue-600 hover:bg-blue-50 rounded-lg transition-colors"
                            title="Edit"
                          >
                            <Edit size={18} />
                          </button>
                          <button 
                            onClick={() => setDeleteConfirmId(t.id)}
                            className="p-2 text-slate-400 hover:text-rose-600 hover:bg-rose-50 rounded-lg transition-colors"
                            title="Hapus"
                          >
                            <Trash2 size={18} />
                          </button>
                        </div>
                      )}
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};

// 4. Report Component
const Report = ({ transactions }) => {
  const [reportYear, setReportYear] = useState(new Date().getFullYear().toString());

  const recapData = useMemo(() => {
    let yearlyIncome = 0;
    let yearlyExpense = 0;
    const monthlyData = Array.from({ length: 12 }, () => ({ income: 0, expense: 0 }));
    const categoryExpense = {};
    const categoryIncome = {};

    transactions.forEach(t => {
      if (t.date.startsWith(reportYear)) {
        const monthIndex = parseInt(t.date.split('-')[1]) - 1;
        
        if (t.type === 'pemasukan') {
          yearlyIncome += t.amount;
          monthlyData[monthIndex].income += t.amount;
          categoryIncome[t.category] = (categoryIncome[t.category] || 0) + t.amount;
        } else {
          yearlyExpense += t.amount;
          monthlyData[monthIndex].expense += t.amount;
          categoryExpense[t.category] = (categoryExpense[t.category] || 0) + t.amount;
        }
      }
    });

    return { yearlyIncome, yearlyExpense, monthlyData, categoryExpense, categoryIncome };
  }, [transactions, reportYear]);

  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <h2 className="text-2xl font-bold text-slate-800">Laporan Keuangan</h2>
        <select 
          value={reportYear}
          onChange={(e) => setReportYear(e.target.value)}
          className="px-4 py-2 border border-slate-200 rounded-lg font-medium text-slate-700 outline-none focus:ring-2 focus:ring-blue-500"
        >
          {Array.from({length: 5}, (_, i) => new Date().getFullYear() - i).map(year => (
            <option key={year} value={year}>Tahun {year}</option>
          ))}
        </select>
      </div>

      {/* Recap Cards */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <div className="bg-emerald-600 rounded-2xl p-6 text-white shadow-md">
          <p className="text-emerald-100 font-medium mb-2">Total Pemasukan ({reportYear})</p>
          <h3 className="text-3xl font-bold">{formatIDR(recapData.yearlyIncome)}</h3>
        </div>
        <div className="bg-rose-600 rounded-2xl p-6 text-white shadow-md">
          <p className="text-rose-100 font-medium mb-2">Total Pengeluaran ({reportYear})</p>
          <h3 className="text-3xl font-bold">{formatIDR(recapData.yearlyExpense)}</h3>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* Monthly Recap Table */}
        <div className="lg:col-span-2 bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
          <h3 className="text-lg font-bold text-slate-800 mb-4 flex items-center gap-2">
            <PieChart size={20} className="text-blue-500"/> Arus Kas Bulanan
          </h3>
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm">
              <thead>
                <tr className="border-b border-slate-100 text-slate-500">
                  <th className="py-3 font-medium">Bulan</th>
                  <th className="py-3 font-medium text-right text-emerald-600">Pemasukan</th>
                  <th className="py-3 font-medium text-right text-rose-600">Pengeluaran</th>
                  <th className="py-3 font-medium text-right text-blue-600">Surplus / Defisit</th>
                </tr>
              </thead>
              <tbody>
                {months.map((month, idx) => {
                  const income = recapData.monthlyData[idx].income;
                  const expense = recapData.monthlyData[idx].expense;
                  const net = income - expense;
                  if(income === 0 && expense === 0) return null; // Sembunyikan bulan kosong
                  return (
                    <tr key={month} className="border-b border-slate-50">
                      <td className="py-3 font-medium text-slate-700">{month}</td>
                      <td className="py-3 text-right">{formatIDR(income)}</td>
                      <td className="py-3 text-right">{formatIDR(expense)}</td>
                      <td className={`py-3 text-right font-bold ${net >= 0 ? 'text-blue-600' : 'text-orange-600'}`}>
                        {formatIDR(net)}
                      </td>
                    </tr>
                  )
                })}
              </tbody>
            </table>
          </div>
        </div>

        {/* Expense by Category */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
          <h3 className="text-lg font-bold text-slate-800 mb-4">Pengeluaran by Kategori</h3>
          {Object.keys(recapData.categoryExpense).length === 0 ? (
            <div className="text-center text-slate-400 py-8">
              <AlertCircle size={32} className="mx-auto mb-2 opacity-50" />
              <p>Belum ada data pengeluaran</p>
            </div>
          ) : (
            <div className="space-y-4">
              {Object.entries(recapData.categoryExpense)
                .sort((a, b) => b[1] - a[1])
                .map(([cat, amount]) => {
                  const percentage = Math.round((amount / recapData.yearlyExpense) * 100);
                  return (
                    <div key={cat}>
                      <div className="flex justify-between text-sm mb-1">
                        <span className="font-medium text-slate-700">{cat}</span>
                        <span className="text-slate-600">{formatIDR(amount)}</span>
                      </div>
                      <div className="w-full bg-slate-100 rounded-full h-2">
                        <div 
                          className="bg-rose-500 h-2 rounded-full" 
                          style={{ width: `${percentage}%` }}
                        ></div>
                      </div>
                    </div>
                  )
              })}
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

// --- MAIN APP COMPONENT ---
export default function App() {
  const [activeTab, setActiveTab] = useState('dashboard');
  const [transactions, setTransactions] = useState([]);
  const [isLoaded, setIsLoaded] = useState(false);

  // Load from LocalStorage
  useEffect(() => {
    const savedData = localStorage.getItem('bukukas_data');
    if (savedData) {
      try {
        setTransactions(JSON.parse(savedData));
      } catch (e) {
        console.error("Failed to parse data");
      }
    }
    setIsLoaded(true);
  }, []);

  // Save to LocalStorage
  useEffect(() => {
    if (isLoaded) {
      localStorage.setItem('bukukas_data', JSON.stringify(transactions));
    }
  }, [transactions, isLoaded]);

  const tabs = [
    { id: 'dashboard', label: 'Dashboard', icon: LayoutDashboard },
    { id: 'transactions', label: 'Transaksi', icon: Receipt },
    { id: 'reports', label: 'Laporan', icon: PieChart },
  ];

  if (!isLoaded) return <div className="min-h-screen bg-slate-50 flex items-center justify-center">Loading...</div>;

  return (
    <div className="min-h-screen bg-slate-50 font-sans text-slate-900 selection:bg-blue-100">
      
      {/* Sidebar/Top Navigation */}
      <nav className="bg-white border-b border-slate-200 sticky top-0 z-10 shadow-sm">
        <div className="max-w-6xl mx-auto px-4">
          <div className="flex justify-between h-16">
            <div className="flex items-center gap-2 text-blue-600">
              <Wallet size={28} className="stroke-[2.5]" />
              <span className="text-xl font-black tracking-tight">BukuKas<span className="text-slate-800">Pro</span></span>
            </div>
            
            {/* Desktop Tabs */}
            <div className="hidden sm:flex space-x-1 items-center">
              {tabs.map(tab => {
                const Icon = tab.icon;
                const isActive = activeTab === tab.id;
                return (
                  <button
                    key={tab.id}
                    onClick={() => setActiveTab(tab.id)}
                    className={`px-4 py-2 rounded-lg flex items-center gap-2 font-medium transition-all ${
                      isActive 
                        ? 'bg-blue-50 text-blue-700' 
                        : 'text-slate-500 hover:bg-slate-50 hover:text-slate-700'
                    }`}
                  >
                    <Icon size={18} />
                    {tab.label}
                  </button>
                )
              })}
            </div>
          </div>
        </div>
      </nav>

      {/* Main Content Area */}
      <main className="max-w-6xl mx-auto px-4 py-8 pb-24 sm:pb-8">
        {activeTab === 'dashboard' && <Dashboard transactions={transactions} />}
        {activeTab === 'transactions' && <TransactionManager transactions={transactions} setTransactions={setTransactions} />}
        {activeTab === 'reports' && <Report transactions={transactions} />}
      </main>

      {/* Mobile Bottom Navigation */}
      <div className="sm:hidden fixed bottom-0 left-0 right-0 bg-white border-t border-slate-200 z-20 pb-safe">
        <div className="flex justify-around items-center h-16">
          {tabs.map(tab => {
            const Icon = tab.icon;
            const isActive = activeTab === tab.id;
            return (
              <button
                key={tab.id}
                onClick={() => setActiveTab(tab.id)}
                className={`flex flex-col items-center justify-center w-full h-full space-y-1 transition-colors ${
                  isActive ? 'text-blue-600' : 'text-slate-400 hover:text-slate-600'
                }`}
              >
                <Icon size={22} className={isActive ? 'stroke-[2.5]' : ''} />
                <span className="text-[10px] font-semibold">{tab.label}</span>
              </button>
            )
          })}
        </div>
      </div>
      
    </div>
  );
}
